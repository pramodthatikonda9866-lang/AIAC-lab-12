code:
def binary_search(arr, target):
    low = 0
    high = len(arr) - 1
    while low <= high:
        mid = (low + high) // 2
        if arr[mid] == target:
            return mid
        elif arr[mid] < target:
            low = mid + 1
        else:
            high = mid - 1
    return -1

code:import time
def linear_search(arr, target):
    comparisons = 0
    start = time.perf_counter()
    for i in range(len(arr)):
        comparisons += 1
        if arr[i] == target:
            elapsed = time.perf_counter() - start
            return i, comparisons, elapsed
    elapsed = time.perf_counter() - start
    return -1, comparisons, elapsed
def binary_search(arr, target):
    low = 0
    high = len(arr) - 1
    comparisons = 0
    start = time.perf_counter()
    while low <= high:
        mid = (low + high) // 2
        comparisons += 1
        if arr[mid] == target:
            elapsed = time.perf_counter() - start
            return mid, comparisons, elapsed
        elif arr[mid] < target:
            low = mid + 1
        else:
            high = mid - 1
    elapsed = time.perf_counter() - start
    return -1, comparisons, elapsed
